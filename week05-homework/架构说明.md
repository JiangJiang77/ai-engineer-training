六、Retry Manager

核心逻辑：

def handle_retry(agent_name, state, error):

    retry_count = state["retry_count"].get(agent_name, 0)

    state["logs"].append(
        f"{agent_name} failed: {error}"
    )

    if retry_count < 2:

        state["retry_count"][agent_name] = retry_count + 1

        state["logs"].append(
            f"Retry {agent_name} ({retry_count+1}/2)"
        )

        return "retry"

    if has_fallback(agent_name):

        state["logs"].append(
            f"Switch to fallback agent"
        )

        return "fallback"

    state["need_user_input"] = True
    state["user_message"] = "需要补充信息"

    return "ask_user"
七、Agent Executor（关键）

统一封装 Agent 执行。

def execute_agent(agent, validator, state, name):

    try:

        result = agent(state)

        if not validator(result):

            raise Exception("validation failed")

        state.update(result)

        return state

    except Exception as e:

        action = handle_retry(name, state, str(e))

        if action == "retry":
            return execute_agent(agent, validator, state, name)

        if action == "fallback":
            fallback = get_fallback_agent(name)
            return execute_agent(fallback, validator, state, name)

        return state
八、LangGraph 构建
from langgraph.graph import StateGraph

builder = StateGraph(WriterState)

builder.add_node("research", research_node)
builder.add_node("write", writing_node)
builder.add_node("review", review_node)
builder.add_node("polish", polishing_node)

builder.set_entry_point("research")

builder.add_edge("research", "write")
builder.add_edge("write", "review")
builder.add_edge("review", "polish")

graph = builder.compile()
